function setCellOption(type, code) { if (!editSelectedCellData) return; if (!editCurrentActivityId) { alert("Silakan klik Tampilkan dulu!"); return; } const { sid, day } = editSelectedCellData; if (!editAttendanceData[sid]) editAttendanceData[sid] = {}; let present = false; let note = ''; if (type === 'present') { present = true; note = ''; } else if (type === 'note') { present = false; note = code; } else { present = false; note = ''; } editAttendanceData[sid][day] = { present, note }; // Tutup modal & render UI dulu (optimistic) closeCellEditor(); renderEditTable(); const dayStr = String(day).padStart(2, '0'); const docId = `${editCurrentYear}-${editCurrentMonth}-${dayStr}_${editCurrentActivityId}`; const docRef = db.collection('attendance').doc(docId); console.log(`[Save Cell] docId="${docId}" sid="${sid}" day=${day} present=${present} note="${note}"`); // ✅ PERBAIKAN: Nested object syntax (sama seperti index.html) const updateData = { records: { [sid]: { present: present, note: note, updatedAt: firebase.firestore.FieldValue.serverTimestamp() } } }; docRef.set(updateData, { merge: true }) .then(() => { console.log(`✓ Presensi ${sid} tgl ${day} tersimpan`); showToast('Tersimpan ✓'); // Verify return docRef.get(); }) .then(doc => { if (doc.exists) { const data = doc.data(); const saved = data.records && data.records[sid]; if (saved && saved.present === present && saved.note === note) { console.log(`[Verify OK] ${docId}/records.${sid}:`, saved); } else { console.warn(`[Verify FAIL] Saved:`, saved, `Expected:`, { present, note }); showToast('⚠️ Periksa Console!'); } } else { console.warn(`[Verify FAIL] Doc ${docId} tidak ada!`); } }) .catch(err => { console.error(`✗ Gagal menyimpan:`, err); showToast('Gagal: ' + err.message); }); }
